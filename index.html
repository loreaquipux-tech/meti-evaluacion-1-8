const NOMBRE_HOJA = 'Resultados METI'; const MAX_INTENTOS = 2; function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(20000); const datos = JSON.parse(e.postData.contents); const nombre = String(datos.nombre || '').trim(); const correo = String(datos.correo || '').trim().toLowerCase(); const puntuacion = Number(datos.puntuacion); if (!nombre || nombre.length > 150 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(correo) || correo.length > 254 || !Number.isInteger(puntuacion) || puntuacion < 0 || puntuacion > 3) { throw new Error('Datos invalidos'); } const libro = SpreadsheetApp.getActiveSpreadsheet(); let hoja = libro.getSheetByName(NOMBRE_HOJA); if (!hoja) hoja = libro.insertSheet(NOMBRE_HOJA); const cabeceras = ['Fecha y hora', 'Estudiante', 'Correo', 'Puntuación', 'Total posible', 'Intento', 'Mejor puntuación']; hoja.getRange(1, 1, 1, cabeceras.length).setValues([cabeceras]); const ultima = hoja.getLastRow(); const registros = ultima > 1 ? hoja.getRange(2, 1, ultima - 1, 7).getValues() : []; let intentos = 0; let mejor = -1; registros.forEach((fila, i) => { const correoFila = String(fila[2] || '').trim().toLowerCase(); if (correoFila !== correo) return; intentos++; const nota = Number(fila[3]); if (Number.isFinite(nota)) mejor = Math.max(mejor, nota); // Completa el numero de intento de registros previos. if (!fila[5]) hoja.getRange(i + 2, 6).setValue(intentos); }); if (intentos >= MAX_INTENTOS) { return respuesta({ok: false, motivo: 'limite_de_intentos', intentos: intentos}); } const textoSeguro = valor => /^[=+\-@]/.test(valor) ? "'" + valor : valor; const mejorActual = Math.max(mejor, puntuacion); hoja.appendRow([new Date(), textoSeguro(nombre), textoSeguro(correo), puntuacion, 3, intentos + 1, mejorActual]); return respuesta({ok: true, intento: intentos + 1, mejor: mejorActual}); } catch (error) { return respuesta({ok: false, motivo: 'error_de_registro'}); } finally { if (lock.hasLock()) lock.releaseLock(); } } function respuesta(objeto) { return ContentService.createTextOutput(JSON.stringify(objeto)) .setMimeType(ContentService.MimeType.JSON); }
